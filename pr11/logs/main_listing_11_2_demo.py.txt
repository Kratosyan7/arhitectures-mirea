# ========== ПАРАМЕТРЫ ВАРИАНТА (ИЗМЕНЯЕМАЯ ЧАСТЬ) ==========
VARIANT_NUM = 1                    # Номер варианта (1-30)
FUNCTION_TYPE = 'sin'              # Тип функции: 'sin', 'x2', 'exp'
FUNCTION_NAME = 'k × sin(x)'       # Название функции для вывода
K = 10.0                           # Параметр функции
A = 4.0                            # Нижняя граница интегрирования

N = 10_000_000                     # Число разбиений
THREADS_LIST = [2, 4, 8]           # Потоки/процессы для теста (значения используются и для MPI)
OUTPUT_DIR = 'results'             # Директория для результатов
# ============================================================


def estimate_sequential_fraction(metrics):
    """
    Оценивает долю последовательного кода из экспериментальных данных.
    Использует закон Амдала: α = (1/S - 1/p) / (1 - 1/p)
    """
    if not metrics:
        return 0.1

    last_metric = metrics[-1]
    p = last_metric['processors']
    S = last_metric['speedup']

    if p > 1 and S > 0:
        alpha = (1/S - 1/p) / (1 - 1/p)
        return max(0.01, min(0.3, alpha))
    else:
        return 0.1

        results_openmp.append(result)

                print(f"  Время (локальный тест): {time_mpi:.4f}s")
